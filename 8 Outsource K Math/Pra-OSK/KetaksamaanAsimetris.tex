\section{Soal-Soal}
\begin{problem}
Jika $a$ dan $b$ adalah bilangan real positif yang memenuhi $ab = 1$, nilai minimum dari $a^2 + 2b^2$ adalah ... dan tercapai saat $a = ...$ serta $b = ...$
\end{problem}
\begin{problem}
Jika $x$ adalah bilangan real positif, nilai minimum dari \[\frac{x^2 + 1}{x + 1}\] adalah ...
\end{problem}
\begin{problem}
$x, y, z$ adalah tiga bilangan real positif dengan $xy + yz + zx = 5$. Nilai minimal dari $$3x^2 + 3y^2  + z^2$$ adalah ...
\end{problem}
\begin{problem}
Untuk setiap bilangan real positif $a,b,c$, buktikan bahwa \[x^4 + y^4 + z^2 \geq \sqrt{8}xyz\]
\end{problem}
\begin{problem}
$a$ dan $b$ adalah bilangan real dengan $a\neq 0$. Nilai minimal dari $$a^2 + b^2 + \frac{1}{a^2} + \frac{b}{a}$$ adalah ... dan tercapai saat ...
\end{problem}
\begin{problem}
$a, b, c$ adalah bilangan real positif yang memenuhi $a + b + c  = 1$. Nilai minimum dari $$\frac{a + b}{abc}$$ adalah ...
\end{problem}
\begin{problem}
Diberikan bilangan real positif $a$ dan $b$ yang memenuhi $\sqrt{a} - \sqrt{b} = 20$. Nilai maksimum dari $a - 5b$ adalah ...
\end{problem}
\begin{problem}
Misal $a, b, c, d$ adalah bilangan real yang memenuhi persamaan $ab + bc + cd + da = 1$. Tentukan nilai minimal dari \[5a^2 + 4b^2 + 5c^2 + d^2\]
\end{problem}
\begin{problem}
Misal $x, y, z, w$ adalah bilangan real sedemikian hingga $x + 2y + 3z + 4w = 1$. Tentukan nilai minimal dari \[x^2 + y^2 + z^2 + w^2 + (x + y + z + w)^2\]
\end{problem}
\begin{problem}
Jika $x$ adalah bilangan real, nilai maksimal dari $4x - x^4$ adalah ...
\end{problem}
\begin{problem}
Diberikan bilangan real positif $x$. Tentukan nilai minimal dari \[x + \dfrac{1}{4x} + \dfrac{x}{(2x + 1)^2}\]
\end{problem}
