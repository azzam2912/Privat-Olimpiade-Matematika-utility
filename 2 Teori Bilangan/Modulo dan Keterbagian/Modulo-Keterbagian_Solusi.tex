\section{Solusi Soal Latihan}

\subsection{Keterbagian}

\begin{enumerate}
	\item Pernyataan B salah. Ambil $a = b = 2$ dan $c = 2$:
	\[ a \mid c \quad \text{dan} \quad b \mid c \quad \text{benar, tetapi} \quad ab = 4 \nmid 2 = c. \]
	Keempat pernyataan lainnya benar:
	A karena $b \mid bc$ lalu gunakan sifat transitif;
	C dan E langsung dari definisi;
	D karena
	\[ 0 = a \cdot 0. \]

	\item Jumlah digit keempat bilangan itu:
	\[ 2 + 2 = 4, \quad 2 + 1 + 1 + 2 = 6, \quad 2 + 2 + 2 + 2 = 8, \quad 2 + 3 + 3 + 2 = 10. \]
	Tidak ada yang habis dibagi $9$, jadi jawabannya $0$.

	\item Habis dibagi $15$ berarti habis dibagi $3$ dan $5$.
	Digit terakhir harus $0$, sebab $8$ ganjil.
	Jadi $n$ terdiri dari $k$ digit $8$ diikuti satu digit $0$,
	dan jumlah digitnya $8k$ harus habis dibagi $3$.
	Nilai $k$ terkecil adalah $k = 3$, sehingga
	\[ n = 8880 \quad \text{dan} \quad \frac{n}{15} = 592. \]

	\item Jumlah keempat bilangan:
	\[ n + (n+1) + (n+2) + (n+3) = 4n + 6 = 2(2n+3). \]
	Untuk $n = 1$ jumlahnya $10$ dan untuk $n = 2$ jumlahnya $14$, sehingga
	\[ p \mid FPB(10, 14) = 2. \]
	Di sisi lain $2$ selalu membagi $4n + 6$.
	Jadi $p = 2$.

	\item Tulis
	\[ n^2 + 1 = (n+1)(n-1) + 2. \]
	Syaratnya setara dengan
	\[ n + 1 \mid 2, \]
	sehingga
	\[ n + 1 \in \{\pm 1, \pm 2\} \quad \text{dan} \quad n \in \{0, -2, 1, -3\}. \]
	Ada $4$ bilangan bulat yang memenuhi.

	\item Kalikan pembilang dan penyebut dengan $4$:
	\[ 4(x^2 - 2x + 7) = (2x-1)(2x-3) + 25. \]
	Jika $2x - 1 \mid x^2 - 2x + 7$, maka
	\[ 2x - 1 \mid 25. \]
	Konversnya juga benar karena $2x - 1$ ganjil sehingga relatif prima dengan $4$.
	Jadi
	\[ 2x - 1 \in \{\pm 1, \pm 5, \pm 25\}, \]
	yang memberi $6$ nilai $x$.
	Banyak himpunan bagiannya:
	\[ 2^6 = 64. \]

	\item Jika keempat bilangan bersisa sama modulo $N$,
	maka $N$ membagi selisihnya:
	\[ 4165, \quad 2975, \quad \text{dan} \quad 1190. \]
	Hitung $FPB$ ketiganya:
	\begin{align*}
	4165 - 2975 &= 1190, \\
	2975 &= 2 \cdot 1190 + 595, \\
	1190 &= 2 \cdot 595.
	\end{align*}
	Karena $595$ membagi ketiga selisih itu, $N = 595$.

	\item Bagi bersusun:
	\[ \frac{n^2+x}{n+1} = n - 1 + \frac{x+1}{n+1}. \]
	Jadi $n + 1$ harus faktor dari $x + 1$ yang nilainya minimal $2$.
	Tepat dua faktor semacam itu berarti
	\[ x + 1 = p^2 \]
	untuk suatu bilangan prima $p$.
	Nilai terbesar di bawah $1000$ adalah
	\[ x + 1 = 31^2 = 961, \]
	sehingga $x = 960$.

	\item Misalkan $d = n^2 + 2$. Dari $d \mid 2001n + 2$ diperoleh
	\[ d \mid n(2001n+2) - 2001(n^2+2) = 2n - 4002. \]
	Kemudian
	\[ 2(2001n+2) - 2001(2n-4002) = 8008006, \]
	sehingga
	\[ d \mid 8008006 = 2 \cdot 19 \cdot 83 \cdot 2539. \]
	Untuk $n = 2001$ syarat terpenuhi dengan hasil bagi $1$.

	Untuk $n > 2001$, ketaksamaan $n^2 + 2 \le 2(n - 2001)$ tidak punya solusi.

	Untuk $n < 2001$, ketaksamaan $n^2 + 2 \le 2(2001 - n)$ memberi $n \le 62$.
	Pembagi $8008006$ yang di bawah $62^2 + 2 = 3846$ adalah
	\[ 1, 2, 19, 38, 83, 166, 1577, 2539, 3154, \]
	dan yang berbentuk $n^2 + 2$ hanyalah
	\[ 38 = 6^2 + 2 \quad \text{dan} \quad 83 = 9^2 + 2. \]
	Pemeriksaan langsung:
	\[ \frac{2001 \cdot 6 + 2}{38} = 316 \quad \text{dan} \quad \frac{2001 \cdot 9 + 2}{83} = 217. \]
	Jadi $n \in \{6, 9, 2001\}$.

	\item Jumlah kuadrat dan jumlah biasa:
	\begin{align*}
	1^2 + \dots + n^2 &= \frac{n(n+1)(2n+1)}{6}, \\
	1 + \dots + n &= \frac{n(n+1)}{2},
	\end{align*}
	sehingga hasil baginya
	\[ \frac{2n+1}{3}. \]
	Syaratnya
	\[ 3 \mid 2n + 1, \quad \text{yaitu} \quad n \equiv 1 \pmod{3}. \]
	Bilangan $n \le 221$ yang memenuhi ada
	\[ \frac{220-1}{3} + 1 = 74. \]

	\item Aturan $11$: $3a - 2b$ harus habis dibagi $11$.
	Nilai $3a - 2b$ berada di antara $-15$ dan $27$,
	sehingga kemungkinannya
	\[ -11, \quad 0, \quad 11, \quad \text{atau} \quad 22. \]
	Keempat kasus memberi
	\begin{align*}
	3a - 2b = -11 &:\quad (a, b) = (1, 7), \\
	3a - 2b = 0 &:\quad (a, b) = (2, 3), (4, 6), (6, 9), \\
	3a - 2b = 11 &:\quad (a, b) = (5, 2), (7, 5), (9, 8), \\
	3a - 2b = 22 &:\quad (a, b) = (8, 1).
	\end{align*}
	Bilangan terkecil $17171$ dan terbesar $98989$.

	\item Pola digitnya membentuk blok berisi satu, dua, sampai dua belas digit $2$,
	masing-masing diikuti satu digit $d$:
	\[ 2 + 3 + \dots + 13 = 90 \ \text{digit}. \]
	Jadi ada $12$ digit $d$ dan $78$ digit $2$,
	dengan jumlah digit
	\[ 156 + 12d. \]
	Agar habis dibagi $9$ perlu
	\[ 3 + 3d \equiv 0 \pmod{9}, \quad \text{yaitu} \quad d \equiv 2 \pmod{3}. \]
	Karena $d \ne 2$, diperoleh $d = 5$ atau $d = 8$.

	\item Tulis
	\[ n^3 + 100 = (n + 10)(n^2 - 10n + 100) - 900. \]
	Syaratnya setara dengan
	\[ n + 10 \mid 900. \]
	Dengan $n + 10 \ge 11$, pembagi $900$ yang memenuhi ada $19$ buah,
	memberi
	\[ n \in \{2, 5, 8, 10, 15, 20, 26, 35, 40, 50, 65, 80, 90, 140, 170, 215, 290, 440, 890\}. \]

	\item Untuk $n \le 2014$, nilai $\frac{8n}{9999-n}$ positif dan tidak lebih dari
	\[ \frac{16112}{7985} < 2{,}02, \]
	sehingga nilainya $1$ atau $2$.
	Nilai $1$ memberi $n = 1111$, sedangkan nilai $2$ memberi $n = 1999{,}8$.
	Jadi hanya ada $1$ bilangan yang memenuhi.

	\item Tulis $n = k^2 + r$ dengan $0 \le r \le 2k$, sehingga
	\[ \lfloor \sqrt{n} \rfloor = k. \]
	Syarat $k \mid n$ setara dengan
	\[ k \mid r, \quad \text{sehingga} \quad r \in \{0, k, 2k\}. \]
	Jawabannya
	\[ n = k^2, \quad k^2 + k, \quad \text{atau} \quad k^2 + 2k \]
	untuk setiap bilangan asli $k$, dan semuanya memang memenuhi.

	\item Tulis $a_{i+1} = c_i a_i$ dengan $c_i \ge 2$ bulat.
	Jumlahnya:
	\[ a_1(1 + c_1 + c_1c_2 + \dots + c_1\cdots c_9) = 1918 = 2 \cdot 959. \]
	Jika $a_1 = 1$, tulis $S = 1918 = 1 + c_1T_1$:
	\[ c_1T_1 = 1917 = 3^3 \cdot 71 \quad \text{dengan} \quad T_1 \ge 255, \]
	yang memaksa $c_1 = 3$ dan $T_1 = 639$.
	Mengulang cara yang sama ($638 = 2 \cdot 319$, $318 = 2 \cdot 159$, dan seterusnya)
	memberi satu-satunya rantai
	\[ 1, 3, 6, 12, 24, 48, 96, 192, 384, 1152. \]
	Jika $a_1 = 2$, maka $S = 959$ prima dan pelacakan yang sama buntu di tengah jalan.
	Nilai $a_1 \ge 959$ mustahil karena $S \ge 1023$.
	Jadi satu-satunya kemungkinan bilangan kesepuluh adalah $1152$.

	\item Jumlah digit berselang-seling $x_n$ berbentuk $C_a a + C_b b$.
	Blok ke-$k$ (dengan $k$ digit $a$ lalu satu digit $b$) berkontribusi
	$(-1)^{E_k}b$ plus $-(-1)^{E_k}a$ jika $k$ ganjil,
	dengan $E_k$ paritas posisi blok itu dari kanan.
	Karena $D_k = k(k+3)/2$ (banyak digit sampai blok $k$) genap tepat saat
	$k \equiv 0$ atau $1 \pmod{4}$, penjumlahan berpasangan memberi
	\begin{align*}
	C_b = 0 &\iff n \equiv 0 \ \text{atau} \ 2 \pmod{4}, \\
	C_a = 0 &\iff n \equiv 0 \ \text{atau} \ 3 \pmod{4}.
	\end{align*}
	Keduanya nol (nilai keduanya di $\{-1, 0, 1\}$, jadi nol modulo $11$ berarti nol)
	jika dan hanya jika
	\[ 4 \mid n. \]
	Ada
	\[ \lfloor 2020/4 \rfloor = 505 \]
	nilai $n$ yang memenuhi.
\end{enumerate}

\subsection{FPB dan KPK}

\begin{enumerate}[resume]
	\item Kurangkan berulang (algoritma Euclid).
	Selisih kedua bilangan adalah
	\[ 2121212121210 - 1121212121211 = 999999999999, \]
	lalu
	\begin{align*}
	1121212121211 - 999999999999 &= 121212121212, \\
	999999999999 &= 8 \cdot 121212121212 + 30303030303, \\
	121212121212 &= 4 \cdot 30303030303.
	\end{align*}
	Jadi $FPB$nya $30303030303$, dan pecahannya senilai dengan
	\[ \frac{70}{37}. \]

	\item Pembilang dan penyebutnya:
	\begin{align*}
	\frac{104 \cdot 105}{2} &= 5460, \\
	\frac{(3+106) \cdot 104}{2} &= 5668.
	\end{align*}
	Hitung $FPB$nya:
	\begin{align*}
	5668 - 5460 &= 208, \\
	5460 &= 26 \cdot 208 + 52, \\
	208 &= 4 \cdot 52.
	\end{align*}
	Jadi $FPB(5460, 5668) = 52$, sehingga
	\[ a = 105, \quad b = 109, \quad a + b = 214. \]

	\item Tulis $a = 2^x 5^y$ dan $b = 2^z 5^w$ dengan
	\[ x + z = y + w = 10. \]
	Syarat $a \mid b^2$ dan $b \mid a^2$ menjadi
	\[ x \le 2z \quad \text{dan} \quad z \le 2x, \]
	yaitu
	\[ 10 - x \le 2x \quad \text{dan} \quad x \le 2(10 - x), \]
	sehingga
	\[ x \in \{4, 5, 6\}. \]
	Hal yang sama berlaku untuk $y$.
	Jadi ada $3 \cdot 3 = 9$ pasangan.

	\item Algoritma Euclid memberi $FPB(133, 198) = 1$ (lihat contoh di Materi).
	Untuk yang kedua,
	\[ 6436343 = 529 \cdot 12167 \]
	tepat, sehingga
	\[ FPB(6436343, 12167) = 12167. \]

	\item Untuk setiap bilangan prima $p$, pasangan valuasi $(v_p(x), v_p(y))$
	harus bernilai
	\[ \min = v_p(5!) \quad \text{dan} \quad \max = v_p(50!). \]
	Karena $v_p(5!) < v_p(50!)$ untuk semua $15$ bilangan prima sampai $50$,
	masing-masing prima memberi $2$ pilihan (yang mana mendapat nilai min).
	Jadi ada
	\[ 2^{15} = 32768 \]
	pasangan.

	\item Karena $FPB(n! + 1, n!) = 1$, berlaku
	\[ f(n) = FPB(n! + 1, n + 1). \]
	Jika $n + 1$ prima, Wilson memberi $f(n) = n + 1$.

	Jika $n + 1$ komposit dan bukan $4$, maka $(n+1) \mid n!$ sehingga $f(n) = 1$;
	untuk $n + 1 = 4$ diperoleh
	\[ f(3) = FPB(7, 4) = 1 \]
	juga.

	Jadi $f(n) = n + 1$ saat $n + 1$ prima dan $1$ selainnya.
	Prima dari $2$ sampai $31$ berjumlah $160$,
	dan $19$ nilai komposit menyumbang $1$, sehingga totalnya
	\[ 160 + 19 = 179. \]

	\item Misalkan $d_n = FPB(a_n, a_{n+1})$; syaratnya $d_n$ konstan.
	Karena $d_n \mid 2(n+10)$ untuk semua $n$, nilai konstan itu membagi $2$.
	Nilai $a_n$ selalu ganjil (sebab $b$ ganjil), jadi
	\[ d_n = 1 \quad \text{untuk semua} \ n. \]
	Jika $p \mid a_n$ dan $p \mid n + 10$, maka
	\[ a_n \equiv b - 90 \pmod{p}, \]
	sehingga $p \mid b - 90$.
	Agar tidak ada prima semacam itu untuk $n$ manapun, perlu
	\[ |b - 90| \le 1, \quad \text{yaitu} \quad b \in \{89, 90, 91\}. \]
	Yang ganjil terbesar adalah $91$, dan pemeriksaan langsung menunjukkan $b = 91$
	(serta $b = 89$) memang memberi $d_n = 1$ untuk semua $n$.

	\item Tulis $d = FPB(m, n)$, $m = dm'$, $n = dn'$ dengan $FPB(m', n') = 1$.
	Maka $KPK(m, n) = dm'n'$ dan syaratnya menjadi
	\[ 1 + m'n' = m' + n', \]
	yaitu
	\[ (m' - 1)(n' - 1) = 0. \]
	Jadi $m' = 1$ atau $n' = 1$, yang berarti $m \mid n$ atau $n \mid m$.

	\item Tulis syaratnya
	\[ \frac{a^2 + b^2 + a + b}{ab} = k \quad \text{bulat}. \]
	Dengan $d = FPB(a, b)$, $a = da_1$, $b = db_1$:
	\[ d^2(a_1^2 + b_1^2) + d(a_1 + b_1) = kd^2a_1b_1, \]
	sehingga
	\[ a_1 + b_1 = dM \quad \text{dengan} \quad M = ka_1b_1 - a_1^2 - b_1^2 \]
	bulat positif.
	Jadi $d \mid a_1 + b_1$, maka
	\[ d \le a_1 + b_1 = \frac{a+b}{d}, \]
	yaitu
	\[ d^2 \le a + b. \]

	\item Syaratnya setara dengan
	\[ xy = z(y - x). \]
	Tulis $x = hA$, $y = hB$, $z = hC$ dengan $FPB(A, B, C) = 1$,
	sehingga
	\[ AB = C(B - A). \]
	Misalkan $d_0 = FPB(A, B)$; karena $A'B'$ koprima dengan $B' - A'$,
	diperoleh
	\[ B' - A' \mid d_0. \]
	Tulis $d_0 = (B' - A')K$, maka
	\[ C = KA'B'. \]
	Karena $K$ membagi $A$, $B$, $C$ sekaligus, $K = 1$.
	Jadi
	\[ B - A = (B' - A')^2 \quad \text{dan} \quad ABC = (d_0A'B')^2 \]
	keduanya kuadrat, sehingga
	\[ h(y - x) = h^2(B - A) \quad \text{dan} \quad hxyz = h^4ABC \]
	keduanya kuadrat.

	\item Persamaannya homogen berderajat $3$, jadi cukup mengklasifikasikan
	tripel primitif ($FPB(a, b, c) = 1$) lalu mengalikan dengan $t$.

	Untuk setiap prima $p$ dengan valuasi $\alpha$, $\beta$, $\gamma$,
	ketiga syarat keterbagian memaksa valuasi terkecil muncul sedikitnya dua kali
	(jika unik, pertidaksamaan valuasinya langsung kontradiksi).
	Akibatnya $p$ membagi tepat satu atau tepat tiga dari $a, b, c$;
	pada tripel primitif tepat satu, jadi $a, b, c$ saling koprima berpasangan.
	Dengan demikian
	\[ KPK(a^2b, b^2c, c^2a) = (abc)^2 \mid S = a^3 + b^3 + c^3; \]
	tulis
	\[ S = k(abc)^2. \]

	Putar tripel sehingga $a$ terbesar.
	Jika $bc \ge \sqrt{a}$, maka
	\[ k = \frac{a}{b^2c^2} + \frac{b}{a^2c^2} + \frac{c}{a^2b^2} \le 3. \]
	Nilai $k = 3$ memaksa $b^2c^2 = a$ dan $S = 3a^3$, yaitu $(1, 1, 1)$.

	Nilai $k = 2$ memberi $a^2 \mid b^3 + c^3$ dengan hasil bagi sekurangnya $a$,
	sehingga $\max(b, c) = a$ untuk $a \ge 5$ (kontradiksi dengan kekoprimaan),
	dan pemeriksaan $a \le 4$ tidak memberi solusi.

	Nilai $k = 1$ memberi
	\[ b^3 + c^3 = a^2e \quad \text{dengan} \quad e = b^2c^2 - a \ge 1. \]
	Jika $e = 1$, maka
	\[ a + 1 = b^2c^2 \quad \text{dan} \quad a^2 = b^3 + c^3: \]
	bilangan $b, c \ge 2$ akan memaksa $a \ge 15$, lalu $\max(b,c) \ge 5$,
	lalu $a \ge 99$, dan seterusnya tanpa henti (mustahil),
	jadi salah satu dari $b, c$ bernilai $1$, yang memberi $(3, 1, 2)$ dan $(3, 2, 1)$.
	Jika $e \ge 2$, maka
	\[ 2a^2 \le a^2e = b^3 + c^3 \le 2\max(b,c)^3, \]
	jadi $\max(b,c) \ge a^{2/3}$ dan
	\[ \min(b, c) \le \sqrt{3a}/a^{2/3} = \sqrt{3}\,a^{-1/6} < 1 \quad \text{untuk} \ a \ge 28, \]
	sehingga $a \le 27$; pemeriksaan berhingga atas pasangan koprima $bc \le 9$
	(yang memenuhi $a^2 \mid b^3 + c^3$ sekaligus $e = b^2c^2 - a$)
 menunjukkan tidak ada yang memenuhi.

	Jika $bc < \sqrt{a}$, maka
	\[ kb^2c^2 = a + \frac{b^3+c^3}{a^2} \]
	terletak strikt di antara $a$ dan $a + 1$ untuk $a > 4$: mustahil.
	Sisa $a \le 4$ tidak memberi solusi.

	Jadi tripel primitifnya $(1, 1, 1)$ dan semua permutasi $(1, 2, 3)$
	(keduanya diperiksa memenuhi), dan solusi umumnya
	$(t, t, t)$ serta semua permutasi $(t, 2t, 3t)$.
\end{enumerate}

\subsection{Modulo}

\begin{enumerate}[resume]
	\item Karena $10^6 \equiv 1 \pmod{7}$, bagi pangkatnya:
	\[ 999999999 = 6 \cdot 166666666 + 3. \]
	Maka
	\[ 10^{999999999} \equiv 10^3 = 1000 \equiv 6 \pmod{7}. \]
	Sisanya $6$.

	\item Pangkat dari $8$ bersiklus $4$ pada digit terakhir:
	\[ 8, \quad 4, \quad 2, \quad 6. \]
	Eksponen $8^{8^8}$ habis dibagi $4$, sehingga digit terakhirnya $6$.

	\item Sistemnya setara dengan
	\begin{align*}
	N &\equiv -3 \pmod{5}, \\
	N &\equiv -4 \pmod{7}, \\
	N &\equiv -5 \pmod{9}.
	\end{align*}
	Dari $N = 9k + 4$ diperoleh
	\[ 2k + 4 \equiv 3 \pmod{7}, \quad \text{yaitu} \quad k \equiv 3 \pmod{7}. \]
	Tulis $k = 7t + 3$ sehingga
	\[ N = 63t + 31, \]
	lalu
	\[ 3t + 1 \equiv 2 \pmod{5} \quad \text{memberi} \quad t \equiv 2 \pmod{5}. \]
	Nilai terkecil
	\[ N = 63 \cdot 2 + 31 = 157 \]
	dengan jumlah digit $13$.

	\item Tulis $A = 5k + 2$ dan $B = 5m + 3$.
	Maka
	\begin{align*}
	A(A+1) &= 25k^2 + 25k + 6 \equiv 6 \pmod{25}, \\
	5B &= 25m + 15 \equiv 15 \pmod{25}.
	\end{align*}
	Jumlahnya bersisa $21$ modulo $25$.

	\item Modulo $64$, naikkan $5$ berulang kali:
	\begin{align*}
	5^2 = 25, \quad 5^4 \equiv 49, \quad 5^8 \equiv 33, \quad 5^{16} \equiv 1.
	\end{align*}
	Karena $2021 = 16 \cdot 126 + 5$, diperoleh
	\[ 5^{2021} \equiv 5^5 = 3125 \equiv 53. \]
	Demikian pula
	\[ 11^2 \equiv 57, \quad 11^4 \equiv 49, \quad 11^8 \equiv 33, \quad 11^{16} \equiv 1, \]
	dan $2022 = 16 \cdot 126 + 6$ memberi
	\[ 11^{2022} \equiv 11^6 \equiv 49 \cdot 57 \equiv 41. \]
	Jumlahnya
	\[ 53 + 41 = 94 \equiv 30. \]

	\item Modulo $100$:
	\begin{align*}
	43^2 = 1849 &\equiv 49, \\
	43^4 &\equiv 49^2 = 2401 \equiv 1.
	\end{align*}
	Eksponen $43^{43}$ ganjil dengan $43 \equiv 3 \pmod{4}$, sehingga
	\[ 43^{43} \equiv 3 \pmod{4}. \]
	Jadi
	\[ 43^{43^{43}} \equiv 43^3 = 43 \cdot 49 = 2107 \equiv 7. \]

	\item Tulis $2000 = 16 \cdot 125$.

	Modulo $16$: $-2004 \equiv 1900 \equiv -4$ dan $25 \equiv 121 \equiv 9$,
	sehingga
	\[ (-4)^n - (-4)^n + 9^n - 9^n = 0. \]

	Modulo $125$: $-2004 \equiv 121$ dan $1900 \equiv 25$,
	sehingga
	\[ 121^n - 25^n + 25^n - 121^n = 0. \]
	Kedua faktor membagi bentuk itu untuk setiap $n$,
	jadi semua bilangan asli $n$ memenuhi.

	\item Andaikan
	\[ 3^n + 4^n \equiv 0 \pmod{11}. \]
	Karena $4 \cdot 3 \equiv 1 \pmod{11}$, kalikan kedua ruas dengan $3^n$:
	\[ 9^n \equiv -1 \pmod{11}. \]
	Padahal pangkat dari $9$ modulo $11$ bersiklus
	\[ 9, \quad 4, \quad 3, \quad 5, \quad 1, \]
	tidak pernah $10$. Jadi tidak ada bilangan asli $n$ yang memenuhi.

	\item Dari $a^{777} \equiv 77 \pmod{100}$:
	$a$ harus ganjil (jika genap, ruas kiri $0$ modulo $4$ tetapi $77 \equiv 1$)
	dan tidak habis dibagi $5$ (jika iya, ruas kiri $0$ modulo $25$ tetapi $77 \equiv 2$).
	Jadi
	\[ FPB(a, 100) = 1. \]
	Modulo $4$:
	\[ a^{777} \equiv a \equiv 1. \]
	Modulo $25$ (karena $\varphi(25) = 20$ dan $777 = 38 \cdot 20 + 17$):
	\[ a^{777} \equiv a^{17} \equiv 2. \]
	Pemetaan $x \mapsto x^{17}$ bijektif pada $(\mathbb{Z}/25)^\times$
	karena $FPB(17, 20) = 1$.
	Nilai $a = 17$ memenuhi:
	\begin{align*}
	17^2 &\equiv 14, \\
	17^4 &\equiv 21, \\
	17^8 &\equiv 16, \\
	17^{16} &\equiv 6, \\
	17^{17} &\equiv 102 \equiv 2 \pmod{25}.
	\end{align*}
	Jadi $a \equiv 17 \pmod{25}$ dan $a \equiv 1 \pmod{4}$,
	yang memberi $a \equiv 17 \pmod{100}$ secara tunggal.
	Dua digit terakhirnya $17$.

	\item Karena $7^4 = 2401 \equiv 1 \pmod{100}$, bagi pangkatnya:
	\[ 707 = 4 \cdot 176 + 3. \]
	Maka
	\[ 7^{707} \equiv 7^3 = 343 \equiv 43 \pmod{100}. \]
	Dua digit terakhirnya $43$.

	\item Pangkat dari $3$ modulo $10$ bersiklus
	\[ 3, \quad 9, \quad 7, \quad 1. \]
	Karena $13 \equiv 3 \pmod{10}$ dan $14 \equiv 2 \pmod{4}$,
	digit terakhir $13^{14}$ adalah $9$.
	Eksponen $14^{15}$ habis dibagi $4$,
	sehingga digit terakhir $13^{14^{15}}$ adalah $1$.

	\item Hasil kalinya ganjil dan memuat faktor $5$ lebih dari tiga kali,
	jadi bersisa
	\[ 0 \pmod{125}. \]
	Modulo $8$, setiap residu ganjil $1, 3, 5, 7$ muncul tepat $253$ kali,
	sehingga hasil kalinya
	\[ \equiv (1 \cdot 3 \cdot 5 \cdot 7)^{253} \equiv 1 \pmod{8}. \]
	Bilangan yang $0$ modulo $125$ dan $1$ modulo $8$ adalah $625$.

	\item Modulo $100$, pangkat dari $32$ bersiklus $4$:
	\begin{align*}
	32^1 &\equiv 32, \\
	32^2 &\equiv 24, \\
	32^3 &\equiv 68, \\
	32^4 &\equiv 76.
	\end{align*}
	Karena $1032 \equiv 32 \pmod{100}$ dan $1032 \equiv 0 \pmod{4}$,
	dua digit terakhir $1032^{1032}$ adalah $76$.

	Untuk yang kedua, $17^4 \equiv 1 \pmod{40}$ memberi
	\[ 17^{17} \equiv 17 \pmod{40}; \]
	tulis $17^{17} = 40t + 17$.
	Karena $17^{40} \equiv 1 \pmod{100}$ (Euler),
	\[ 17^{17^{17}} \equiv 17^{17} \pmod{100}. \]
	Hitung:
	\begin{align*}
	17^2 &\equiv 89, \\
	17^4 &\equiv 21, \\
	17^8 &\equiv 41, \\
	17^{16} &\equiv 81,
	\end{align*}
	sehingga
	\[ 17^{17} \equiv 81 \cdot 17 = 1377 \equiv 77 \pmod{100}. \]

	\item Bilangan $m$ digit yang seluruh digitnya $d$ bernilai $dR_m$
	dengan $R_m$ repunit $m$ digit.
	Syaratnya $10 < dR_m < 100000$ (jadi $2 \le m \le 5$)
	dan
	\[ dR_m \equiv 1 \pmod{3}. \]
	Karena $R_m \equiv m \pmod{3}$, syaratnya
	\[ dm \equiv 1 \pmod{3}: \]
	$3$ nilai $d$ untuk $m = 2$, tidak ada untuk $m = 3$,
	$3$ nilai untuk $m = 4$, dan $3$ nilai untuk $m = 5$.
	Total $9$ bilangan.

	\item Karena $FPB(b, m) = 1$, $b$ punya invers modulo $m$.
	Misalkan
	\[ u \equiv ab^{-1} \pmod{m}; \]
	maka
	\[ u^x \equiv u^y \equiv 1 \pmod{m}. \]
	Dengan $d = FPB(x, y) = sx + ty$ (Bezout) diperoleh
	\[ u^d = (u^x)^s(u^y)^t \equiv 1 \pmod{m}, \]
	yaitu
	\[ a^d \equiv b^d \pmod{m}. \]

	\item Jika $x^2 \equiv -1 \pmod{p}$ dengan $p$ ganjil,
	maka $x$ berorde $4$ modulo $p$, sehingga
	\[ 4 \mid p - 1. \]
	Konversnya, dari Wilson:
	\[ (p-1)! \equiv \left(\left(\frac{p-1}{2}\right)!\right)^2(-1)^{(p+1)/2} \equiv -1 \pmod{p}. \]
	Untuk $p \equiv 1 \pmod{4}$ nilai $(p+1)/2$ ganjil,
	sehingga $x = \left(\frac{p-1}{2}\right)!$ memenuhi
	\[ x^2 \equiv -1 \pmod{p}. \]
	(Catatan: $p = 2$ pengecualian, $x = 1$ memenuhi tetapi $2 \not\equiv 1 \pmod{4}$.)

	\item Tulis repunitnya
	\[ \frac{10^{p-1} - 1}{9}. \]
	Fermat Kecil memberi
	\[ 10^{p-1} \equiv 1 \pmod{p}, \]
	dan $9$ punya invers modulo $p$ karena $p \ge 7$.
	Jadi $p$ membagi repunit tersebut.

	\item Untuk $p = 2$ pemeriksaan langsung memberi $a$ dan $b$ genap.

	Untuk $p > 2$: andaikan $p \nmid b$, maka
	\[ t \equiv ab^{-1} \pmod{p} \]
	memenuhi
	\[ t^2 + t + 1 \equiv 0, \]
	sehingga $t^3 \equiv 1$ dan $t \not\equiv 1$.
	Jadi $t$ berorde $3$, maka
	\[ 3 \mid p - 1. \]
	Tetapi $p = 3k + 2$ memberi $p - 1 \equiv 1 \pmod{3}$: kontradiksi.
	Jadi $p \mid b$, lalu $p \mid a^2$, lalu $p \mid a$.

	\item Andaikan $n \mid 3^n + 1$ dan ambil $p$ pembagi prima terkecil $n$ ($p$ ganjil).
	Maka
	\[ 3^n \equiv -1 \pmod{p} \quad \text{dan} \quad 3^{2n} \equiv 1 \pmod{p}. \]
	Orde $d$ dari $3$ modulo $p$ memenuhi $d \mid 2n$ tetapi $d \nmid n$,
	sehingga
	\[ v_2(d) = v_2(n) + 1 = 1 \quad \text{(karena $n$ ganjil)}. \]
	Jadi $d/2 \mid n$ dengan $d/2 < p$ (sebab $d \mid p - 1$),
	sehingga $d/2 = 1$ karena $p$ pembagi prima terkecil.
	Maka $d = 2$, yaitu $p \mid 8$: kontradiksi dengan $p$ ganjil.

	\item Misalkan $d$ membagi keduanya dengan $m < n$.
	Tulis $A = a^{2^m}$ dan $B = 2^{2^m}$:
	\[ d \mid A + B \quad \text{dan} \quad d \mid A^{2^k} + B^{2^k} \]
	dengan $k = n - m \ge 1$.
	Karena $A \equiv -B \pmod{d}$ dan $2^k$ genap,
	\[ d \mid 2B^{2^k} = 2^{2^n + 1}. \]
	Jadi $d$ pangkat dari $2$, tetapi kedua bilangan jelas ganjil.
	Jadi $d = 1$.

	\item Tulis $p = kn + 1$. Karena $p > n - 1 \ge 1$,
	$p \nmid n - 1$ sehingga
	\[ p \mid n^2 + n + 1. \]
	Dari $kn + 1 \le n^2 + n + 1$ diperoleh $k \le n + 1$,
	lalu $p \mid n(n + 1 - k)$ dengan $p \nmid n$ memberi
	\[ p \mid n + 1 - k. \]
	Karena $0 \le n + 1 - k < p$, harus $k = n + 1$.
	Jadi
	\[ p = n^2 + n + 1 \quad \text{dan} \quad 4p - 3 = (2n+1)^2. \]

	\item Misalkan $d$ pembagi bersama itu dan $q$ faktor primanya.
	Modulo $q$: $a \equiv -b$ sehingga
	\[ S = \frac{a^p+b^p}{a+b} \equiv p\,a^{p-1} \pmod{q}. \]
	Jika $q \mid a$ maka $q \mid b$, kontradiksi dengan kekoprimaan.
	Jadi $q \nmid a$ dan $q \mid p$, yaitu $q = p$.

	Jadi $d = p^e$; andaikan $e \ge 2$, maka
	\[ S \equiv p\,a^{p-1} \pmod{p^2} \]
	memberi $p \mid a$, lalu $p \mid b$: kontradiksi dengan kekoprimaan.
	Jadi $d \in \{1, p\}$.

	\item Pasangkan $i$ dengan $p - i$.
	Dari
	\[ (p-i)^p \equiv -i^p \pmod{p^2} \]
	diperoleh
	\[ r_{p-i} \equiv -r_i \pmod{p^2}. \]
	Karena $0 < r_i, r_{p-i} < p^2$ (sebab $p \nmid i$), berlaku
	\[ r_i + r_{p-i} = p^2. \]
	Ada $\frac{p-1}{2}$ pasangan, jadi jumlahnya
	\[ \frac{(p-1)p^2}{2}. \]

	\item Cukup menunjukkan $p^e \mid 2^{n!} - 1$ untuk setiap $p^e \parallel n^2 - 1$.
	Karena $n$ genap, $n - 1$ dan $n + 1$ keduanya ganjil dan koprima,
	sehingga $p^e$ membagi salah satunya dan
	\[ p^e \le n + 1. \]
	Misalkan $t$ orde $2$ modulo $p$:
	\[ t \mid p - 1 \le n \quad \text{dan} \quad p^{e-1} \le n. \]
	Keduanya membagi $n!$ dan saling koprima, jadi
	\[ t\,p^{e-1} \mid n!. \]
	LTE memberi
	\[ v_p(2^{t\,p^{e-1}} - 1) \ge e, \]
	sehingga $p^e \mid 2^{n!} - 1$. Gabungkan semua faktor prima dengan CRT.

	\item Andaikan $2^b - 1 \mid 2^a + 1$ dengan $b > 2$.

	Jika $b$ ganjil, orde $2$ modulo setiap faktor prima $p$ dari $2^b - 1$
	membagi $b$ (jadi ganjil) dan $2a$ tetapi bukan $a$: mustahil,
	karena orde ganjil yang membagi $2a$ pasti membagi $a$.

	Jadi $b$ genap; jika $4 \mid b$, maka modulo $3$ memberi $a$ ganjil
	sedangkan modulo $5$ memberi $a$ genap: kontradiksi.

	Jadi $b = 2u$ dengan $u \ge 3$ ganjil.
	Orde $2$ modulo $2^u + 1$ membagi $2u$ tetapi bukan $u$,
	dan tepat sama dengan $2u$: orde $2d' < 2u$ berarti $d' \le u/2$,
	sehingga
	\[ 2^u + 1 \le 2^{2d'} - 1 < 2^u: \quad \text{mustahil}. \]
	Jadi $2u \mid 2a$, yaitu $u \mid a$.
	Tulis $a = uv$: modulo $2^u - 1$ berlaku
	\[ 2^a = (2^u)^v \equiv 1, \]
	tetapi juga $2^a \equiv -1$: kontradiksi karena $2^u - 1 \ge 7$.

	\item Tulis $1989 = 3^2 \cdot 13 \cdot 17$ dan $T_2 = n^n$, $T_3 = n^{T_2}$;
	soalnya
	\[ n^{T_3} - n^{T_2}. \]
	Jika $3 \mid n$, maka $T_2 \ge 27$ sehingga $9$ membagi kedua suku.
	Jika $13 \mid n$ atau $17 \mid n$, kesimpulannya langsung.

	Jika tidak, Euler mereduksi soal menjadi
	\begin{align*}
	6 \mid T_3 - T_2 &\quad (\text{mod } 9), \\
	12 \mid T_3 - T_2 &\quad (\text{mod } 13), \\
	16 \mid T_3 - T_2 &\quad (\text{mod } 17).
	\end{align*}
	Selalu
	\[ T_3 - T_2 = n^n(n^{T_2-n} - 1) \quad \text{dengan} \quad T_2 - n \ \text{genap}. \]
	Faktor $2$ dipenuhi lewat paritas ($n^n$ atau $n^{T_2-n} - 1$ genap);
	faktor $3$ dipenuhi $n^n$ saat $3 \mid n$ dan
	\[ n^{T_2-n} \equiv 1 \pmod{3} \quad \text{saat} \quad 3 \nmid n. \]
	Untuk faktor $16$ (mod $17$): jika $n$ genap ($n \ge 4$) maka $16 \mid n^n$;
	jika $n$ ganjil maka $v_2(T_2 - n) \ge 3$ dan LTE memberi
	\[ v_2(n^{T_2-n} - 1) \ge 3 + 3 - 1 = 5 \ge 4. \]
	Untuk faktor $4$ (mod $13$): jika $n$ genap maka $4 \mid n^n$;
	jika $n$ ganjil maka
	\[ n^{T_2-n} \equiv 1 \pmod{8}. \]

	\item Ambil sebarang prima $p > \max(a, b)$ dan
	\[ n = (a+1)(p-1) + 1. \]
	Maka
	\[ a^n \equiv a \pmod{p} \quad \text{dan} \quad n \equiv -a \pmod{p}, \]
	sehingga
	\[ p \mid a^n + n. \]
	Syarat memberi $p \mid b^n + n$, dan $b^n \equiv b \pmod{p}$,
	sehingga
	\[ b \equiv a \pmod{p}. \]
	Berlaku untuk semua prima besar $p$, jadi $a = b$.

	\item Untuk $a \ge b$:
	\[ x^a - 1 = x^{a-b}(x^b - 1) + (x^{a-b} - 1), \]
	sehingga
	\[ FPB(x^a - 1, x^b - 1) = FPB(x^b - 1, x^{a-b} - 1). \]
	Ini tepat langkah algoritma Euclid pada eksponen,
	yang berakhir di
	\[ x^{FPB(a,b)} - 1. \]

	\item Jika $p \mid a$, ambil $m = p$.

	Jika tidak, CRT memberi $m$ dengan
	\[ m \equiv a \pmod{p} \quad \text{dan} \quad m \equiv 1 \pmod{p-1}. \]
	Maka
	\[ m^m \equiv 1 \pmod{p-1}; \]
	tulis $m^m = 1 + s(p-1)$, sehingga
	\[ m^{m^m} \equiv a^{m^m} = a(a^{p-1})^s \equiv a \pmod{p}. \]
\end{enumerate}

\subsection{Bilangan Prima dan Pangkat}

\begin{enumerate}[resume]
	\item Modulo $3$, ketiga bilangan bersisa
	\[ p, \quad p + 2, \quad p + 1: \]
	mencakup semua sisa. Salah satunya habis dibagi $3$,
	dan karena ketiganya prima, yang habis dibagi $3$ itu harus sama dengan $3$.
	Hanya $p = 3$ yang mungkin, dan $3$, $11$, $19$ semuanya prima.

	\item Faktorisasi Sophie Germain:
	\[ a^4 + 4b^4 = (a^2 - 2ab + 2b^2)(a^2 + 2ab + 2b^2). \]
	Agar hasilnya prima, faktor terkecil harus $1$:
	\[ (a-b)^2 + b^2 = 1, \]
	sehingga $a = b = 1$.
	Faktor terbesarnya $5$, yang prima. Jadi $(a, b) = (1, 1)$.

	\item Jika $m \ge 2$ dan $n \ge 2$, maka
	\[ 6^m + 2^n + 2 \equiv 2 \pmod{4}, \]
	bukan kuadrat. Sisanya dua kasus.

	Untuk $m = 1$:
	\[ 8 + 2^n = k^2. \]
	Nilai $n = 1$ dan $n = 2$ gagal, $n = 3$ memberi $k = 4$,
	dan untuk $n \ge 4$ diperoleh $k$ genap dengan
	\[ (k/2)^2 = 2 + 2^{n-2} \equiv 2 \pmod{4}, \]
	mustahil.

	Untuk $n = 1$:
	\[ 6^m + 4 = k^2, \quad \text{yaitu} \quad (k-2)(k+2) = 2^m 3^m. \]
	Nilai $m = 1$ memberi $10$, bukan kuadrat.
	Untuk $m \ge 2$, $k$ genap dan $k > 2$.
	Karena $k^2 \equiv 1 \pmod{3}$, tepat satu dari $k-2$, $k+2$ habis dibagi $3$.

	Jika $k \equiv 0 \pmod{4}$, kedua faktor bersisa $2$ modulo $4$
	sehingga $m = 2$, tetapi
	\[ 6^2 + 4 = 40 \]
	bukan kuadrat.

	Jika $k \equiv 2 \pmod{4}$, tulis $k - 2 = 2^a 3^b$ dan $k + 2 = 2^c 3^d$
	dengan $a, c \ge 2$, $a + c = m$, dan $\{b, d\} = \{0, m\}$.
	Apabila $b = 0$ dan $c = a$:
	\[ 2^a(3^m - 1) = 4, \]
	mustahil karena $3^m - 1 \ge 8$.
	Apabila $b = 0$ dan $c > a$:
	\[ 2^a = 4 \quad \text{dan} \quad 2^{c-2} 3^m = 2, \]
	mustahil.
	Apabila $b = 0$ dan $c < a$:
	\[ 2^c = 4 \quad \text{dan} \quad 3^m - 2^{m-4} = 1 \quad (m \ge 4), \]
	mustahil karena ruas kirinya minimal $80$.
	Apabila $d = 0$: harus $c > a$, lalu
	\[ 2^a = 4 \quad \text{dan} \quad 2^{m-4} - 3^m = 1, \]
	mustahil karena ruas kirinya negatif.
	Jadi tidak ada solusi dengan $n = 1$.
	Satu-satunya solusi adalah $(m, n) = (1, 3)$.

	\item Jika $n > 2$ genap, maka $3n - 4$ genap dan minimal $8$: komposit.

	Jika $n \ge 3$ ganjil, maka $5n - 3$ genap dan minimal $12$: komposit.

	Untuk $n = 1$ bilangannya negatif.
	Tinggal $n = 2$ yang memberi $2$, $3$, $7$, semuanya prima.

	\item Modulo $6$, bilangan prima $p > 3$ tidak mungkin bersisa
	\[ 0, \quad 2, \quad 3, \quad \text{atau} \quad 4 \]
	(genap atau kelipatan $3$).
	Jadi
	\[ p \equiv 1 \ \text{atau} \ 5 \pmod{6}, \]
	yaitu $p = 6k + 1$ atau $6k + 5 = 6(k+1) - 1$.

	\item Sama seperti contoh di Materi: $p$ ganjil memberi
	\[ 8 \mid p^2 - 1 \]
	(dari dua bilangan genap berurutan $p - 1$, $p + 1$),
	lalu $p \not\equiv 0 \pmod{3}$ memberi
	\[ 3 \mid p^2 - 1, \]
	dan $8$ koprima dengan $3$.

	\item Setiap bilangan di barisan (mulai $11$) berakhiran digit $11$,
	sehingga bersisa
	\[ 3 \pmod{4}. \]
	Kuadrat hanya bersisa $0$ atau $1$ modulo $4$.
	Jadi tidak ada yang kuadrat: jawabannya $0$.

	\item Periksa $n \le 4$:
	\[ 6, \quad 7, \quad 11, \quad 29, \]
	bukan kubik.
	Untuk $n = 5$:
	\[ 120 + 5 = 125 = 5^3. \]
	Untuk $n \ge 6$: $9 \mid n!$ sehingga
	\[ n! + 5 \equiv 5 \pmod{9}, \]
	tetapi kubik modulo $9$ hanya $0$, $1$, atau $8$.
	Jadi satu-satunya jawaban $n = 5$.

	\item Jelas $p, q \ne 2$. Modulo $3$: jika $p \equiv 0$ maka $p + 6 = 9$,
	jika $p \equiv 2$ maka $3 \mid p + 10$; jadi
	\[ p \equiv 1 \pmod{3}. \]
	Serupa, $q = 3$ atau $q \equiv 1 \pmod{3}$.
	Jika $q \equiv 1$, maka $p + q + 1 \equiv 0 \pmod{3}$ dan lebih dari $3$: mustahil.
	Jadi $q = 3$, dan $p \equiv 1 \pmod{3}$ dengan $p + 4$, $p + 6$, $p + 10$ prima.
	Pemeriksaan prima $1$ modulo $3$ di bawah $100$ memberi
	\[ p \in \{7, 13, 37, 97\}, \]
	yang semuanya diperiksa memenuhi.

	\item Pasangkan suku ke-$i$ dan ke-$(p-i)$:
	\[ \frac{1}{i} + \frac{1}{p-i} = \frac{p}{i(p-i)}. \]
	Jumlahnya
	\[ p \cdot \frac{A}{B} \]
	dengan $B$ produk semua $i(p-i)$, yang tidak habis dibagi $p$.
	Dalam bentuk paling sederhana, faktor $p$ di pembilang tidak bisa tercoret,
	jadi $p \mid a$.

	\item Jika $a$ ganjil, $a^n + 1$ genap dan lebih dari $2$: komposit.
	Jadi $a$ genap.

	Jika $n$ punya faktor ganjil $t > 1$, tulis $n = mt$:
	\[ a^n + 1 = (a^m)^t + 1 \]
	habis dibagi $a^m + 1 > 1$: komposit.
	Jadi $n$ tidak punya faktor ganjil lebih dari $1$, yaitu $n = 2^k$.

	\item Karena $a - 1 \mid a^n - 1$, keprimaan memaksa
	\[ a - 1 = 1, \quad \text{yaitu} \quad a = 2. \]
	Jika $n = de$ komposit, maka
	\[ 2^n - 1 = (2^d)^e - 1 \]
	habis dibagi $2^d - 1 > 1$.
	Jadi $n$ prima.

	\item Untuk setiap $i = 1, \ldots, n$, pilih prima berbeda $p_i > n$.
	CRT memberi $x$ dengan
	\[ x + i \equiv p_i \pmod{p_i^2} \quad \text{untuk semua} \ i, \]
	sehingga
	\[ v_{p_i}(x + i) = 1. \]
	Setiap $x + i$ punya faktor prima berpangkat tepat satu,
	jadi bukan bilangan berpangkat sempurna.
	(Ambil wakil $x$ yang cukup besar agar semua positif.)

	\item Ya. Tulis $a_i = \prod_p p^{\alpha_{i,p}}$ dan $P = \prod_i a_i$.
	Pilih prima berbeda $q_1, \ldots, q_k$.
	Untuk setiap $p \mid P$, CRT memberi $\beta_p$ dengan
	\[ \beta_p \equiv -\alpha_{i,p} \pmod{q_i} \quad \text{untuk semua} \ i. \]
	Maka $b = \prod_p p^{\beta_p}$ membuat $ba_i$ berpangkat $q_i$ sempurna.

	\item Nilai $p = 2$ memberi $347$, bukan kubik.
	Untuk $p$ ganjil, tulis $k^3 = p^2 + 343 > 343$, jadi $k \ge 8$ dan
	\[ p^2 = (k - 7)(k^2 + 7k + 49) \]
	dengan kedua faktor positif.
	Karena $p^2$ hanya berfaktor $(1, p^2)$ dan $(p, p)$,
	dan $k - 7 < k^2 + 7k + 49$,
	kasus $(p, p)$ memberi
	\[ k^2 + 6k + 56 = 0, \]
	yang mustahil.
	Jadi $k - 7 = 1$, $k = 8$, dan
	\[ p^2 = 169, \quad \text{yaitu} \quad p = 13. \]
	Periksa:
	\[ 13^2 + 343 = 512 = 8^3. \]
\end{enumerate}

\section{Solusi Soal Ujian}

\begin{enumerate}
	\item (Soal Wajib) Untuk setiap $n$, tunjukkan $2$, $3$, dan $5$ membagi $n^5 - n$.

	Paritas $n^5$ sama dengan $n$, jadi $n^5 - n$ genap.
	Modulo $3$: karena $n^3 \equiv n$,
	\[ n^5 = n^3 \cdot n^2 \equiv n \cdot n^2 = n^3 \equiv n \pmod{3}. \]
	Modulo $5$: Fermat Kecil memberi $n^5 \equiv n \pmod{5}$.
	Jadi
	\[ 30 \mid n^5 - n \quad \text{untuk semua} \ n. \]
	Untuk $n = 2$ nilainya tepat $30$, sehingga $30$ adalah pembagi terbesar.

	\item (Soal Wajib) Karena
	\[ 2(3p+25) - 3(2p-5) = 65, \]
	syaratnya berakibat
	\[ 2p - 5 \mid 65. \]
	Pembagi ganjil dari $65$ memberi
	\[ p \in \{3, 5, 9, 35\} \]
	(setelah membuang $p = 2$ yang membuat hasil baginya negatif
	dan nilai $p$ yang tidak positif).
	Keempatnya diperiksa memenuhi dengan hasil bagi
	\[ 34, \quad 8, \quad 4, \quad 2. \]

	\item (Soal Wajib) Untuk setiap $n$:
	\begin{align*}
	2 \mid n^7 - n &\quad \text{(paritas)}, \\
	3 \mid n^7 - n &\quad \text{(karena $n^3 \equiv n$ memberi $n^7 \equiv n$)}, \\
	7 \mid n^7 - n &\quad \text{(Fermat Kecil)}.
	\end{align*}
	Jadi
	\[ 42 \mid n^7 - n \quad \text{untuk semua} \ n. \]
	Nilai terbesar $k$ yang selalu membagi adalah pembagi bersama terbesar
	dari semua nilai $n^7 - n$:
	untuk $n = 2$ nilainya $126$ dan $n = 3$ nilainya $2184$,
	dengan
	\[ FPB(126, 2184) = 42. \]
	Jadi $k$ yang memenuhi tepat pembagi positif dari $42$,
	yang banyaknya $8$.

	\item (Soal Pilihan) Tulis $S = 1^n + \dots + m^n$.
	Untuk $n$ ganjil, pasangkan suku ke-$k$ dengan suku ke-$(m+1-k)$:
	\[ 2S = \sum_{k=1}^{m} \bigl(k^n + (m+1-k)^n\bigr), \]
	dan setiap pasangan habis dibagi $m + 1$.
	Dengan pasangan $k$ dan $m - k$ diperoleh pula
	\[ m \mid 2S. \]
	Karena $m$ dan $m + 1$ relatif prima,
	\[ m(m+1) \mid 2S, \quad \text{yaitu} \quad \frac{m(m+1)}{2} \mid S. \]
	Jadi setiap $n$ ganjil valid, khususnya $2013$.

	Sebaliknya, untuk $n$ genap ambil $m = 2$:
	$\frac{1 + 2^n}{3}$ bulat hanya jika
	\[ 2^n \equiv 2 \pmod{3}, \]
	yaitu $n$ ganjil. Jadi semua $n$ genap tidak valid:
	tak hingga banyaknya.

	\item (Soal Pilihan) Karena $209 = 11 \cdot 19$, banyaknya $a$ yang relatif prima:
	\[ \varphi(209) = 10 \cdot 18 = 180. \]
	Dari jumlah itu, yang memenuhi $a^2 \equiv 1 \pmod{209}$ adalah gabungan
	\[ a \equiv \pm 1 \pmod{11} \quad \text{dan} \quad a \equiv \pm 1 \pmod{19}: \]
	tepat $4$ bilangan menurut CRT.
	Jadi yang memenuhi syarat ada
	\[ 180 - 4 = 176. \]
\end{enumerate}
