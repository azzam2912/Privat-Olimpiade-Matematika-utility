\section{Phantom Point}

\begin{definition*}[Phantom point]
Misalkan kita diminta membuktikan bahwa suatu titik $X$ punya sifat $\mathcal{P}$, dan sifat $\mathcal{P}$ itu sendiri sudah cukup untuk menentukan sebuah titik secara tunggal. Alih alih menyerang $X$ langsung, definisikan titik baru $X'$ yang memenuhi $\mathcal{P}$, lalu buktikan $X' = X$. Titik $X'$ inilah yang disebut \emph{phantom point} atau titik bayangan.
\end{definition*}

Kunci teknik ini ada pada ketunggalan \textit{(uniqueness)}. Dua garis yang tidak sejajar berpotongan tepat di satu titik, sebuah garis memotong lingkaran paling banyak di dua titik, dan sebuah busur punya tepat satu titik tengah. Begitu $X$ dan $X'$ terbukti memenuhi syarat yang menentukan satu titik saja, keduanya wajib berimpit (\textit{coincide}).

\begin{lemma*}[Pencerminan titik tinggi terhadap sisi]
Diberikan $\triangle ABC$ dengan titik tinggi $H$. Pencerminan $H$ terhadap garis $BC$ terletak pada lingkaran luar $\triangle ABC$.
\end{lemma*}

\begin{proof}
Misalkan $D$ kaki garis tinggi dari $A$, dan definisikan $H'$ sebagai pencerminan $H$ terhadap $BC$. Kita akan menunjukkan $ABH'C$ siklis, sehingga $H'$ berimpit dengan titik potong kedua garis $AD$ dengan lingkaran luar. Untuk sementara anggap $\triangle ABC$ lancip, supaya $H$ berada di dalam segitiga.

Dari $\angle HBC = 90^\circ - \angle C$ dan $\angle HCB = 90^\circ - \angle B$ kita peroleh
\[
\angle BHC = 180^\circ - \angle HBC - \angle HCB = \angle B + \angle C = 180^\circ - \angle A.
\]
Pencerminan mempertahankan (\textit{preserve}) besar sudut, jadi $\angle BH'C = \angle BHC = 180^\circ - \angle A$. Akibatnya $\angle BAC + \angle BH'C = 180^\circ$, yang berarti $ABH'C$ siklis.

Sekarang $H'$ terletak pada garis $AD$ (karena $H$ dan $H'$ simetris terhadap $BC$ dan $AD \perp BC$) sekaligus pada lingkaran luar. Garis $AD$ memotong lingkaran luar di $A$ dan satu titik lain saja, dan $H' \ne A$. Jadi $H'$ tepat titik potong kedua tersebut.
\end{proof}

Perhatikan alur berpikirnya. Kita tidak pernah menghitung apa pun tentang titik potong kedua garis $AD$ dengan lingkaran luar. Kita membuat titik lainnya, membuktikan ia siklis, lalu memakai \textit{uniqueness} untuk mebuktikan. Hal yang sama muncul di \S\ref{sec:talibusur} untuk membuktikan kolinearitas $T$, $K$, $M$.

\section{Sudut Berarah (\textit{Directed Angles})}
\label{sec:sudut-berarah}

Bukti lemma tadi sebenernya cuma berlaku kalau $\triangle ABC$ lancip, walau nggak pernah dibilang begitu. Coba $\angle A$ tumpul: $H$ pindah ke luar segitiga, $\angle BHC$ jadi $\angle A$ (bukan $180^\circ - \angle A$), dan langkah $\angle BAC + \angle BH'C = 180^\circ$ nggak berlaku lagi, padahal lemmanya sendiri tetap benar. Ini pas banget dijadikan alasan buat kenalan sama sudut berarah, karena dengan itu kita nggak perlu pisah kasus lancip vs tumpul sama sekali.

\begin{definition*}[Sudut berarah]
Untuk dua garis $\ell, m$, notasi $\measuredangle(\ell, m)$ adalah rotasi dari $\ell$ ke $m$, modulo $180^\circ$. Untuk tiga titik,
\[
\measuredangle ABC := \measuredangle(BA, BC).
\]
Bedanya dari $\angle ABC$ biasa: (i) $BA, BC$ dibaca sebagai garis penuh, bukan sinar, dan (ii) hasilnya modulo $180^\circ$, jadi $\measuredangle ABC = -\measuredangle ABC + 180^\circ$ dan $90^\circ = -90^\circ$.
\end{definition*}

\subsection{Properti}

\begin{enumerate}
  \item \textbf{Antisimetri.} $\measuredangle ABC = -\measuredangle CBA$.
  \item \textbf{Penjumlahan berantai.} $\measuredangle(\ell, m) + \measuredangle(m, n) = \measuredangle(\ell, n)$, yakni $\measuredangle ABC + \measuredangle CBD = \measuredangle ABD$.
  \item \textbf{Segaris.} $A, B, C$ segaris $\iff \measuredangle ABC = 0$.
  \item \textbf{Sejajar/tegak lurus.} $\ell \parallel m \iff \measuredangle(\ell,m) = 0$; \quad $\ell \perp m \iff \measuredangle(\ell,m) = 90^\circ$.
  \item \textbf{Jumlah sudut segitiga.} $\measuredangle ABC + \measuredangle BCA + \measuredangle CAB = 0$ (bukan $180^\circ$, karena $180^\circ \equiv 0$).
  \item \textbf{Konsiklis.} Untuk $A, B, X, Y$ berbeda, tanpa tiga segaris,
  \[
  A, B, X, Y \text{ konsiklis} \iff \measuredangle XAY = \measuredangle XBY.
  \]
  Satu baris ini merangkum sudut keliling sebusur sekaligus segiempat talibusur.
  \item \textbf{Sudut singgung.} $t$ menyinggung lingkaran luar $\triangle ABC$ di $A \implies \measuredangle(t, AB) = \measuredangle ACB$.
  \item \textbf{Pencerminan.} $A', B', C'$ pencerminan $A, B, C$ terhadap garis yang sama $\implies \measuredangle A'B'C' = -\measuredangle ABC$.
\end{enumerate}

\subsection{Yang Tidak Boleh Dilakukan}

\begin{itemize}
  \item \textbf{Membagi dua.} Dari $2x = 2y$ (mod $180^\circ$) hanya boleh disimpulkan $x = y$ atau $x = y + 90^\circ$. Garis bagi, titik pusat lingkaran dalam, dan Incenter-Excenter Lemma (\S\ref{sec:incenter-excenter}) di luar jangkauan.
  \item \textbf{Sudut pusat.} Sudut pusat $= 2 \times$ sudut keliling berlaku modulo $360^\circ$, tidak selamat dipaksa masuk modulo $180^\circ$.
  \item \textbf{Panjang.} Begitu argumen menyentuh aturan sinus, rasio kesebangunan, atau segitiga sama kaki, kembali ke sudut biasa.
  \item \textbf{Kata ``di antara''.} Letak relatif titik dan busur besar/kecil tak terlihat oleh $\measuredangle$; kesimpulan tentang letak butuh argumen terpisah.
  \item \textbf{Menulis $180^\circ - \theta$.} Lawan $\theta$ adalah $-\theta$, bukan $180^\circ - \theta$.
\end{itemize}

\subsection{Lemma Pencerminan Titik Tinggi, Sekali Lagi}

\begin{proof}[Bukti ulang dengan sudut berarah]
Karena $BH \perp AC$ dan $CH \perp AB$,
\[
\measuredangle BHC = \measuredangle(HB, AC) + \measuredangle(AC, AB) + \measuredangle(AB, HC)
        = 90^\circ + \measuredangle CAB + 90^\circ = \measuredangle CAB.
\]
$H'$ pencerminan $H$ terhadap $BC$, dengan $B, C$ tetap, jadi
\[
\measuredangle BH'C = -\measuredangle BHC = -\measuredangle CAB = \measuredangle BAC,
\]
dan kriteria konsiklis memberi $A, B, H', C$ satu lingkaran, yaitu lingkaran luar $\triangle ABC$. Sisanya seperti bukti pertama: $H'$ ada pada garis $AD$ karena pencerminan terhadap $BC$ memetakan garis $AD \perp BC$ ke dirinya sendiri, dan $AD$ memotong lingkaran luar di $A$ serta satu titik lain, yaitu $H'$.
\end{proof}

Nggak ada lagi $180^\circ - \angle A$, nggak ada asumsi $H$ di dalam segitiga, dan nggak ada kasus yang perlu dipisah.

\subsection{Sudut Isogonal: $\measuredangle BAH = \measuredangle OAC$}

\begin{theorem*}
Pada $\triangle ABC$ dengan titik tinggi $H$ dan titik pusat lingkaran luar $O$, garis $AH$ dan garis $AO$ simetris terhadap garis bagi $\angle A$. Dalam sudut berarah, $\measuredangle BAH = \measuredangle OAC$.
\end{theorem*}

\begin{proof}
Misalkan $D$ kaki garis tinggi dari $A$, sehingga $AH$ dan $AD$ adalah garis yang sama. Pada $\triangle ABD$ berlaku $\measuredangle ABD + \measuredangle BDA + \measuredangle DAB = 0$ dengan $\measuredangle BDA = 90^\circ$. Karena $D$ pada garis $BC$, kita punya $\measuredangle ABD = \measuredangle ABC$, jadi
\[
\measuredangle BAH = \measuredangle BAD = -\measuredangle DAB = \measuredangle ABC + 90^\circ.
\]

Untuk ruas satunya, tarik garis singgung $t$ pada lingkaran luar di titik $A$. Aturan sudut singgung memberi $\measuredangle(t, AC) = \measuredangle ABC$, sedangkan $OA \perp t$ berarti $\measuredangle(AO, t) = 90^\circ$. Penjumlahan berantai menutup perhitungannya:
\[
\measuredangle OAC = \measuredangle(AO, t) + \measuredangle(t, AC) = 90^\circ + \measuredangle ABC.
\]
Kedua ruas bernilai sama, jadi $\measuredangle BAH = \measuredangle OAC$.
\end{proof}

Perhatikan, buktinya nggak pernah pakai $\triangle AOC$ sama kaki atau sudut pusat, dua hal yang emang dilarang di sudut berarah. Garis singgung di $A$ yang gantiin keduanya. Trik ini worth diinget: tiap ketemu $O$ di angle chasing, coba ganti $O$ dengan garis singgungnya.

Salah satu akibatnya: garis bagi $\angle A$ otomatis jadi garis bagi $\angle HAO$ juga, dan $AH = AO$ persis kalau $\angle A = 60^\circ$.

Kedua pasang sisi berhadapan pada $BHCA'$ sejajar, jadi ia jajar genjang, dan diagonalnya $BC$, $HA'$ saling membagi dua sama panjang di $M$.
\end{proof}

\subsection{Incenter-Excenter Lemma}
\label{sec:incenter-excenter}

\begin{theorem*}[Incenter-Excenter Lemma]
Misalkan $I$ titik pusat lingkaran dalam $\triangle ABC$ dan $I_A$ titik pusat lingkaran singgung luar yang menyinggung sisi $BC$. Jika garis bagi $\angle A$ memotong lingkaran luar lagi di $M$, maka
\[
MB = MI = MC = MI_A.
\]
Dengan kata lain, $M$ adalah pusat lingkaran yang melalui $B$, $I$, $C$, dan $I_A$.
\end{theorem*}

  \draw[gray] (B) -- (I) -- (C);
  \draw[gray,dashed] (B) -- (Ia) -- (C);

  \titik{A,B,C,I,M,Ia}
  \node[above]       at (A)  {$A$};

  \node[above right] at (I)  {$I$};
  \node[right]       at (M)  {$M$};
  \node[below]       at (Ia) {$I_A$};
\end{tikzpicture}
\caption{Titik tengah busur $BC$ menjadi pusat satu lingkaran yang memuat $B$, $I$, $C$, dan $I_A$.}
\end{figure}

\begin{proof}
Di sini kita balik pakai sudut biasa. Seluruh argumennya soal garis bagi dan panjang ruas, dua hal yang emang nggak dikenal $\measuredangle$.

Tulis $\angle A = 2\alpha$, $\angle B = 2\beta$, $\angle C = 2\gamma$. Karena $AM$ garis bagi $\angle A$, titik $M$ adalah titik tengah busur $BC$ yang tidak memuat $A$, sehingga $\angle MBC = \angle MAC = \alpha$. Maka
\[
\angle MBI = \angle MBC + \angle CBI = \alpha + \beta.
\]
Di sisi lain, $\angle MIB$ adalah sudut luar $\triangle ABI$ di titik $I$, jadi
\[
\angle MIB = \angle IAB + \angle IBA = \alpha + \beta.
\]
Kedua sudut sama, jadi $\triangle MBI$ sama kaki dengan $MB = MI$. Argumen simetris memberi $MC = MI$.

Untuk $I_A$, ingat bahwa $BI$ adalah garis bagi dalam $\angle B$ sedangkan $BI_A$ garis bagi luarnya, sehingga $\angle IBI_A = 90^\circ$. Titik $I$, $M$, $I_A$ segaris karena ketiganya ada pada garis bagi $\angle A$. Jadi $\triangle IBI_A$ siku siku di $B$ dengan diameter $II_A$, dan $M$ pada diameter itu memenuhi $MB = MI$. Titik tengah diameter adalah pusat lingkaran yang berjarak sama ke $B$ dan ke $I$, jadi $M$ titik tengah $II_A$ dan $MI_A = MI = MB$.
\end{proof}

\section{Homoteti (Homothety) / Dilatasi}

Homoteti (dilatasi, dalam istilah kurikulum sekolah) adalah transformasi yang mengubah ukuran sebuah bangun tanpa mengubah bentuknya, ditentukan oleh dua data:
\begin{enumerate}
  \item \textbf{Titik pusat} $O$, satu satunya titik yang tidak bergerak, \textit{fixed point}.
  \item \textbf{Faktor skala} $k$, bilangan real tak nol yang sekaligus menentukan besar perubahan ukuran dan arah pemetaannya.
\end{enumerate}

Homoteti dengan pusat $O$ dan faktor $k$ memetakan setiap titik $P$ ke $P'$ dengan
\[
\vec{OP'} = k \cdot \vec{OP}.
\]

Tanda $k$ menentukan letak bayangan: $k>0$ bayangan sesinar dengan aslinya dari $O$ (membesar jika $k>1$, mengecil jika $0<k<1$), sedangkan $k<0$ bayangan menyeberang ke sisi lain $O$ dan bangunnya terbalik.

\subsection{Lingkaran yang Menyinggung Tali Busur}
\label{sec:talibusur}

\begin{theorem*}[Lemma Archimedes]
Diberikan lingkaran $\Omega$ dengan tali busur $AB$. Misalkan $\omega$ lingkaran yang menyinggung $AB$ di $K$ dan menyinggung $\Omega$ dari dalam di $T$. Misalkan $M$ titik tengah busur $AB$ yang tidak memuat $T$. Maka $T$, $K$, $M$ segaris, dan
\[
MK \cdot MT = MB^2 = MA^2.
\]
\end{theorem*}

\caption{Homoteti berpusat $T$ memetakan $\omega$ ke $\Omega$ dan membawa $K$ ke $M$.}

\begin{proof}
Kedua lingkaran bersinggungan dalam di $T$, jadi pusat $\omega$, pusat $\Omega$, dan $T$ segaris. Akibatnya ada homoteti $h$ berpusat $T$ dengan faktor positif yang memetakan $\omega$ ke $\Omega$.

Pakai phantom point. Definisikan $M' = h(K)$, yaitu titik potong kedua garis $TK$ dengan $\Omega$. Homoteti memetakan garis singgung $\omega$ di $K$ ke garis singgung $\Omega$ di $M'$, dan kedua garis itu sejajar. Garis singgung $\omega$ di $K$ tidak lain adalah $AB$, jadi garis singgung $\Omega$ di $M'$ sejajar $AB$. Hanya ada dua titik pada $\Omega$ dengan garis singgung sejajar $AB$, yaitu kedua titik tengah busur $AB$. Karena $M'$ berada di sisi $AB$ yang berseberangan dengan $T$, titik $M'$ adalah titik tengah busur yang tidak memuat $T$. Jadi $M' = M$, dan $T$, $K$, $M$ segaris.

Bagian panjangnya kembali memakai sudut biasa, sebab yang dicari adalah perbandingan sisi. Karena $K$ pada ruas $AB$, garis $BK$ dan garis $BA$ berimpit, jadi $\angle MBK = \angle MBA$. Sudut $\angle MBA$ menghadap busur $MA$ sedangkan $\angle MTB$ menghadap busur $MB$, dan kedua busur itu sama panjang karena $M$ titik tengah busur $AB$. Jadi $\angle MBK = \angle MTB$. Sudut di puncak $M$ dipakai bersama oleh $\triangle MKB$ dan $\triangle MBT$, sehingga $\triangle MKB \sim \triangle MBT$. Perbandingan sisinya memberi $MK/MB = MB/MT$, yaitu $MK \cdot MT = MB^2$. Ruas kiri itu tidak lain kuasa titik $M$ terhadap $\omega$.
\end{proof}

\subsection{Lingkaran Singgung Luar}

Lingkaran dalam dan $A$-excircle sama sama menyinggung kedua sisi sudut $A$, jadi ada homoteti berpusat $A$ yang memetakan yang satu ke yang lain. Itu sebabnya konfigurasi excircle enak dibaca dengan mata homoteti.

\begin{definition*}
$A$-excircle dari $\triangle ABC$ adalah lingkaran yang menyinggung sisi $BC$ serta perpanjangan sisi $AB$ dan $AC$. Pusatnya disebut $A$-excenter dan dinotasikan $I_A$. Analogi definisi untuk sudut $B$ dan $C$.
\end{definition*}

\begin{lemma*}[Panjang garis singgung ke excircle]
Misalkan $B_1$ dan $C_1$ titik singgung $A$-excircle pada perpanjangan $AB$ dan $AC$. Jika $s$ menyatakan setengah keliling $\triangle ABC$, maka
\[
AB_1 = AC_1 = s.
\]
\end{lemma*}

\begin{proof}
Panjang dua garis singgung dari satu titik ke sebuah lingkaran selalu sama, jadi $AB_1 = AC_1$. Misalkan $X$ titik singgung $A$-excircle dengan $BC$. Dengan alasan yang sama, $BB_1 = BX$ dan $CC_1 = CX$. Maka
\begin{align*}
  AB_1 + AC_1 &= (AB + BB_1) + (AC + CC_1) \\
              &= (AB + BX) + (AC + CX) \\
              &= AB + AC + BC = 2s.
\end{align*}
Karena kedua panjang itu sama dan jumlahnya $2s$, masing masing bernilai $s$.
\end{proof}

Dari sini didapat akibat yang sering dipakai: $BX = s - c$, $CX = s - b$, dan titik singgung lingkaran dalam pada $BC$ merupakan pencerminan $X$ terhadap titik tengah $BC$.

\subsection{Kuasa Titik (\textit{Power of a Point - P.O.P.})}
\begin{definition*}
Kuasa titik $P$ terhadap lingkaran $\omega$ berpusat $O$ dan berjari jari $r$ adalah
\[
\operatorname{Pow}_\omega(P) = OP^2 - r^2.
\]
Nilainya positif kalau $P$ di luar $\omega$, nol kalau $P$ pada $\omega$, dan negatif kalau $P$ di dalam $\omega$.
\end{definition*}

\begin{theorem*}
Jika sebuah garis melalui $P$ memotong $\omega$ di $X$ dan $Y$, maka
\[
PX \cdot PY = \left| \operatorname{Pow}_\omega(P) \right|.
\]
Khususnya, kalau $P$ di luar $\omega$ dan $PT$ menyinggung $\omega$ di $T$, maka $PT^2 = \operatorname{Pow}_\omega(P)$.
\end{theorem*}

\begin{theorem*}[Kebalikan kuasa titik]
Misalkan $A$, $B$, $X$, $Y$ empat titik berbeda sehingga garis $AB$ dan garis $XY$ berpotongan di $P$. Jika
\[
PA \cdot PB = PX \cdot PY,
\]
dan $P$ berada di dalam kedua ruas $AB$ dan $XY$ sekaligus, atau di luar keduanya sekaligus, maka $A$, $B$, $X$, $Y$ konsiklis.
\end{theorem*}
\subsection{Sumbu Radikal (\textit{Radical Axis})}

Sumbu radikal $\omega_1, \omega_2$ (berpusat $O_1, O_2$) adalah tempat kedudukan titik dengan $\operatorname{Pow}_{\omega_1}(P) = \operatorname{Pow}_{\omega_2}(P)$, berupa garis lurus tegak lurus $O_1O_2$. Kalau kedua lingkaran berpotongan di $A, B$, sumbu radikalnya persis garis $AB$.

\[
A, B, C, D \text{ konsiklis} \iff P \in \text{sumbu radikal}(\omega_1, \omega_2).
\]
